import fs from 'node:fs';import dotenv from 'dotenv';import {spawn,spawnSync} from 'node:child_process';import {randomUUID} from 'node:crypto';import {chromium} from '@playwright/test';import {createClient} from '@supabase/supabase-js';import pg from 'pg';import net from 'node:net';
const env=dotenv.parse(fs.readFileSync('.env.local'));const fixture=JSON.parse(fs.readFileSync('.cache/provider-fixtures.json'));const credentials=JSON.parse(fs.readFileSync('.cache/provider-fixture-credentials.json'));let base;
const results=[];let stage='startup';let browser;let server;let db;let adminDb;
function check(label,value){stage=label;if(!value)throw Error(label);results.push({check:label,result:'PASS'});console.log(label+': PASS');}
try{
 const probe=net.createServer();await new Promise(r=>probe.listen(0,'127.0.0.1',r));const port=probe.address().port;await new Promise(r=>probe.close(r));base='http://127.0.0.1:'+port;
 server=spawn(process.execPath,['node_modules/next/dist/bin/next','start','--hostname','127.0.0.1','--port',String(port)],{env:{...process.env,...env},windowsHide:true,stdio:['ignore','pipe','pipe']});const log=fs.createWriteStream('.cache/local-verification-server.log');server.stdout.pipe(log);server.stderr.pipe(log);
 const deadline=Date.now()+30000;while(true){if(server.exitCode!==null)throw Error('Verification server exited');try{const r=await fetch(base+'/api/health');if(r.ok)break;}catch{}if(Date.now()>deadline)throw Error('Local server unavailable');await new Promise(r=>setTimeout(r,200));}
 db=new pg.Client({connectionString:env.DATABASE_URL,ssl:{rejectUnauthorized:true,ca:fs.readFileSync(env.DATABASE_CA_CERT_PATH,'utf8')},connectionTimeoutMillis:8000});await db.connect();check('Restricted runtime TLS',db.connection.stream.encrypted&&db.connection.stream.authorized);
 browser=await chromium.launch({headless:true});const a=await browser.newContext({viewport:{width:393,height:851}});const page=await a.newPage();
 stage='Farmer browser sign-in';await page.goto(base+'/sign-in');await page.getByLabel('Email address').fill(credentials[0].email);await page.getByLabel('Password',{exact:true}).fill(credentials[0].password);await page.getByRole('button',{name:'Sign in',exact:true}).click();await page.waitForURL('**/workspace',{timeout:20000});await page.getByRole('heading',{name:'Welcome to MyFarm'}).waitFor();check('Real mobile sign-in and workspace',true);
 check('Secure HttpOnly auth cookies',(await a.cookies()).filter(c=>c.name.startsWith('sb-')).every(c=>c.httpOnly&&c.secure&&c.sameSite==='Lax'));

 const b=await browser.newContext();const adminPage=await b.newPage();stage='Administrator browser sign-in';await adminPage.goto(base+'/sign-in');await adminPage.getByLabel('Email address').fill(credentials[1].email);await adminPage.getByLabel('Password',{exact:true}).fill(credentials[1].password);await adminPage.getByRole('button',{name:'Sign in',exact:true}).click();await adminPage.waitForURL('**/workspace',{timeout:20000});
 await adminPage.goto(base+'/admin?scope='+fixture.tenants[1]);await adminPage.getByRole('heading',{name:'Administrator workspace'}).waitFor();check('Assigned administrator workspace',true);
 await adminPage.goto(base+'/agent?scope='+fixture.tenants[2]);await adminPage.getByRole('heading',{name:'Agent workspace'}).waitFor();check('Assigned agent workspace',true);
 check('Administrator has no global tenant bypass',(await b.request.get(base+'/api/v1/scopes/'+fixture.tenants[0])).status()===403);
 const path=base+'/api/v1/scopes/'+fixture.tenants[0]+'/files/'+fixture.objects[0];const requestId=randomUUID();const options={headers:{origin:base},data:{requestId}};
 const grants=await Promise.all([a.request.post(path,options),a.request.post(path,options)]);console.log(JSON.stringify({privateGrantStatuses:grants.map(r=>r.status())}));if(grants[0].status()===403){const diagnostic=await a.request.post(path,{headers:{origin:base.replace('127.0.0.1','localhost')},data:{requestId:randomUUID()}});console.log(JSON.stringify({normalizedInternalOriginStatus:diagnostic.status()}));}check('Concurrent private grants succeed',grants.every(r=>r.status()===200));const grant=await grants[0].json();check('Short-lived private grant',grant.expiresInSeconds===60);const download=await fetch(grant.url);check('Private signed file downloads',download.status===200&&(await download.text())==='MyFarm Phase 1 temporary verification');
 const count=(await db.query('SELECT count(*)::int AS count FROM "AuditEvent" WHERE "tenantId"=$1 AND "requestId"=$2',[fixture.tenants[0],requestId])).rows[0].count;check('Concurrent grant appends exactly one audit',count===1);
 check('Audit retry payload collision denied',(await a.request.post(base+'/api/v1/scopes/'+fixture.tenants[0]+'/files/'+randomUUID(),options)).status()===409);
 check('Cross-tenant private grant denied',(await b.request.post(path,{headers:{origin:base},data:{requestId:randomUUID()}})).status()===403);
 const anon=createClient(env.NEXT_PUBLIC_SUPABASE_URL,env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,{auth:{persistSession:false,autoRefreshToken:false}});check('Anonymous storage signing denied',!!(await anon.storage.from('myfarm-private').createSignedUrl(fixture.tenants[0]+'/'+fixture.objects[0],60)).error);
 const second=createClient(env.NEXT_PUBLIC_SUPABASE_URL,env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,{auth:{persistSession:false,autoRefreshToken:false}});const secondLogin=await second.auth.signInWithPassword(credentials[1]);check('Second real provider identity',!secondLogin.error);check('Direct storage cross-tenant denial',!!(await second.storage.from('myfarm-private').createSignedUrl(fixture.tenants[0]+'/'+fixture.objects[0],60)).error);
 const owner=createClient(env.NEXT_PUBLIC_SUPABASE_URL,env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,{auth:{persistSession:false,autoRefreshToken:false}});const ownerLogin=await owner.auth.signInWithPassword(credentials[0]);check('Owner real provider identity',!ownerLogin.error);const token=ownerLogin.data.session.access_token;
 const claims=await owner.auth.getClaims();const sessionId=claims.data.claims.session_id;check('Provider session confirmed active',(await db.query('SELECT myfarm_private.myfarm_session_is_active($1::uuid,$2::uuid) AS active',[sessionId,fixture.users[0].authSubject])).rows[0].active);
 const adminEnv=dotenv.parse(fs.readFileSync('.cache/phase01-provider-admin.env'));adminDb=new pg.Client({connectionString:adminEnv.DATABASE_URL,ssl:{rejectUnauthorized:true,ca:fs.readFileSync(env.DATABASE_CA_CERT_PATH,'utf8')},connectionTimeoutMillis:8000});await adminDb.connect();
 await adminDb.query('UPDATE "Membership" SET status=\'REVOKED\' WHERE "userId"=$1 AND "organizationId"=$2',[fixture.users[0].actorId,fixture.tenants[0]]);
 check('Revoked membership API denied',(await a.request.get(base+'/api/v1/scopes/'+fixture.tenants[0])).status()===403);check('Revoked membership direct storage denied',!!(await owner.storage.from('myfarm-private').createSignedUrl(fixture.tenants[0]+'/'+fixture.objects[0],60)).error);
 await adminDb.query('UPDATE "Membership" SET status=\'ACTIVE\' WHERE "userId"=$1 AND "organizationId"=$2',[fixture.users[0].actorId,fixture.tenants[0]]);
 const service=createClient(env.NEXT_PUBLIC_SUPABASE_URL,env.SUPABASE_SERVICE_ROLE_KEY,{auth:{persistSession:false,autoRefreshToken:false}});const link=await service.auth.admin.generateLink({type:'recovery',email:credentials[0].email});check('Provider recovery token generated without email',!link.error&&link.data.properties.verification_type==='recovery');
 const out=await owner.auth.signOut({scope:'global'});check('Provider global sign-out',!out.error);
 check('Provider session revocation confirmed',!(await db.query('SELECT myfarm_private.myfarm_session_is_active($1::uuid,$2::uuid) AS active',[sessionId,fixture.users[0].authSubject])).rows[0].active);
 const revoked=createClient(env.NEXT_PUBLIC_SUPABASE_URL,env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,{global:{headers:{Authorization:'Bearer '+token}},auth:{persistSession:false,autoRefreshToken:false}});check('Revoked JWT cannot obtain new storage grant',!!(await revoked.storage.from('myfarm-private').createSignedUrl(fixture.tenants[0]+'/'+fixture.objects[0],60)).error);
 check('Revoked browser session API denied',(await a.request.get(base+'/api/v1/session')).status()===401);
 await page.goto(base+'/workspace');await page.waitForURL('**/sign-in*');check('Revoked browser redirected to sign-in',true);
 await second.auth.signOut({scope:'global'});await a.close();await b.close();
}catch(e){results.push({check:stage,result:'FAIL',errorCode:e.code??e.name});console.log(JSON.stringify({failedCheck:stage,errorCode:e.code??e.name}));process.exitCode=1;}finally{if(adminDb)await adminDb.end().catch(()=>{});if(db)await db.end().catch(()=>{});if(browser)await browser.close();if(server&&server.exitCode===null){if(process.platform==='win32')spawnSync('taskkill',['/PID',String(server.pid),'/T','/F'],{windowsHide:true,stdio:'ignore'});else server.kill();}fs.writeFileSync('docs/reports/evidence/phase-01-local-verification/provider-tests.json',JSON.stringify({result:process.exitCode?'FAIL':'PASS',results},null,2));}
