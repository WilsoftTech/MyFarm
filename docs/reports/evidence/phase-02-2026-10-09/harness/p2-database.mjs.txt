const results = [];
function step(name, command, args, { cwd = checkout, env = {}, shell = true, expect = 0 } = {}) {
  const r = spawnSync(shell ? process.env.ComSpec : command, shell ? ["/d", "/s", "/c", command] : args, { cwd, env: { ...process.env, ...env }, encoding: "utf8", maxBuffer: 64 * 1024 * 1024, windowsHide: true });
  const out = (r.stdout ?? "") + (r.stderr ?? "");
  writeFileSync(path.join(logDir, name + ".log"), "$ " + (shell ? command : command + " " + args.join(" ")) + "\n" + out + "\nexit=" + r.status + "\n");
  results.push({ step: name, exitCode: r.status, result: r.status === expect ? "PASS" : "FAIL" }); console.log(name, r.status);
  return out;
}
