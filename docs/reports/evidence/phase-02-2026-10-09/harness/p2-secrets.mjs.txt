import fs from 'node:fs'; import path from 'node:path'; import { execSync } from 'node:child_process'; import { env, adminEnv } from './p2-common.mjs';
const secrets = { serviceRoleKey: env.SUPABASE_SERVICE_ROLE_KEY, runtimePassword: decodeURIComponent(new URL(env.DATABASE_URL).password), adminPassword: decodeURIComponent(new URL(adminEnv.DATABASE_URL).password), directUrl: env.DIRECT_URL, testUserPassword: env.MYFARM_TEST_PASSWORD, vercelOidc: env.VERCEL_OIDC_TOKEN };
const walk = d => fs.existsSync(d) ? fs.readdirSync(d, { withFileTypes: true }).flatMap(e => e.isDirectory() ? walk(path.join(d, e.name)) : [path.join(d, e.name)]) : [];
const scan = (label, files) => { const hits = []; for (const f of files) { let t; try { t = fs.readFileSync(f, 'utf8'); } catch { continue; } for (const [k, v] of Object.entries(secrets)) if (v && v.length > 8 && t.includes(v)) hits.push({ file: f, secret: k }); } return { scope: label, files: files.length, hits }; };
const tracked = execSync('git diff --name-only e4ea43b..HEAD', { encoding: 'utf8' }).trim().split('\n').filter(f => fs.existsSync(f));
const history = execSync('git log -p e4ea43b..HEAD', { encoding: 'utf8', maxBuffer: 256 * 1024 * 1024 });
const out = [
  scan('client bundle (.next/static, dev-config build)', walk('.next/static')),
  scan('files changed on branch since Phase1 head', tracked),
  scan('Phase2 evidence folder', walk('docs/reports/evidence/phase-02-2026-10-09')),
  { scope: 'branch commit history (git log -p)', hits: Object.entries(secrets).filter(([, v]) => v && v.length > 8 && history.includes(v)).map(([k]) => k) },
  { scope: 'client bundle references to server-only env names', hits: walk('.next/static').filter(f => /SUPABASE_SERVICE_ROLE_KEY|DATABASE_URL|DIRECT_URL/.test(fs.readFileSync(f, 'utf8'))) },
];
console.log(JSON.stringify({ result: out.every(o => o.hits.length === 0) ? 'PASS' : 'FAIL', scans: out }, null, 1));
